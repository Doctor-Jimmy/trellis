\begin{definition}
For an increasing enumeration $x$ and $n\in\mathbb N$, define
$\mathsf{PowerShiftIter}(n,x)$ recursively by
\[
\mathsf{PowerShiftIter}(0,x)=x,qquad
\mathsf{PowerShiftIter}(n+1,x)=mathsf{ShiftMap}igl(
\mathsf{PowerShiftIter}(n,x)\bigr).
\]
This is the $n$-fold shift orbit used by the copied games.
\end{definition}
