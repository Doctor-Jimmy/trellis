\begin{definition}[Rado order]
For increasing pairs $p=(m,n)$ and $q=(m',n')$, put
\[
p\leq_{\mathsf{Rado}}q
\quad\Longleftrightarrow\quad
(m=m'\ \text{and}\ n\leq n')\ \text{or}\ n<m'.
\]
This is exactly the paper's order on $[\omega]^2$, with each two-element
set written in increasing order.
\end{definition}
