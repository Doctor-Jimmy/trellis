\begin{definition}[Domination on subsets]
For a relation $r$ on $Q$ and subsets $P,T\subseteq Q$, say that $P$
is dominated by $T$, and write $P\preceq_r T$, when
\[
\forall p\in P\ \exists q\in T\quad r(p,q).
\]
The relation $\mathsf{SetDomination}(r)$ on $\mathcal P(Q)$ is this
domination relation.  For a quasi-order it is the first-level powerset
quasi-order used in the paper.
\end{definition}
