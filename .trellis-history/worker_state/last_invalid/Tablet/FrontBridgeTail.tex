\begin{definition}[Finite bridge tail]
For a finite set $u\subseteq\mathbb N$, define
\[
  \mathsf{FrontBridgeTail}(u)=
  \{k\in u\mid \text{some }n\in u\text{ satisfies }n<k\}.
\]
Thus, when $u$ is nonempty, this is $u$ with its least element removed;
it is the finite set seen after applying the shift to the increasing
enumeration of $u$.
\end{definition}
