\begin{lemma}[The terminal atom stabilizes under shifting]
If $h$ is bad for $\mathsf{PowerRel}(r)$, then for every $x$ there are
$k,q,q'$ such that
\[
L_{0,k}=\mathsf{atom}(q),\qquad
L_{1,k}=\mathsf{atom}(q'),\qquad
\mathsf{PowerCopiedLeft}_r(h,\mathsf{ShiftMap}(x),0,k)
=\mathsf{atom}(q').
\]
\end{lemma}
\begin{proof}
Apply \noderef{PowerCopiedFiniteTerminal} to $h$, $hbad$, and $x$.  It
provides $k\in\mathbb N$ and atoms $q,q'\in Q$ such that, writing
$L_{n,k}=\mathsf{PowerCopiedLeft}_r(h,x,n,k)$,
\[
L_{0,k}=\mathsf{atom}(q),\qquad
L_{1,k}=\mathsf{atom}(q'),
\]
and, in particular, records both terminal atom equations for this same
$k,q,q'$.  Now apply \noderef{PowerCopiedShiftIdentity} with $n=0$ and
this $k$.  The resulting equality is
\[
\mathsf{PowerCopiedLeft}_r(h,\mathsf{ShiftMap}(x),0,k)
=\mathsf{PowerCopiedLeft}_r(h,x,0+1,k)
=\mathsf{PowerCopiedLeft}_r(h,x,1,k)=L_{1,k}.
\]
By the second terminal equation above, the right-hand side is exactly
$\mathsf{atom}(q')$, which proves the third required equality.
\end{proof}
