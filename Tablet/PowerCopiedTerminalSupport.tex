\begin{lemma}[The terminal atom lies in the original support]
If $h$ is bad for $\mathsf{PowerRel}(r)$, then for every $x$ there are
$k,q,q'$ such that the two adjacent copied values are
$\mathsf{atom}(q)$ and $\mathsf{atom}(q')$ and
\[
q\in\mathsf{PowerSupport}(h(x)).
\]
\end{lemma}
\begin{proof}
Choose $k,q,q'$ from the termination statement
\noderef{PowerCopiedTerminates} for row zero.  The support containment
\noderef{PowerCopiedSupport} gives
\[
\mathsf{PowerSupport}(\mathsf{atom}(q))
\subseteq \mathsf{PowerSupport}(h(x)).
\]
The atom clause of the support definition \noderef{PowerSupport} says that
$q\in\mathsf{PowerSupport}(\mathsf{atom}(q))$.  Transitivity of inclusion
therefore gives the displayed membership, while the two atom equations are
the chosen terminal witness.
\end{proof}
