\begin{lemma}[Super-sequence perfectness gives base perfectness]
If $f:F\to Q$ is perfect, then there is a base multi-sequence
$h:\mathsf{BaseIncSeq}(X)\to Q$ which is locally constant and perfect, and
which satisfies the extension value equation
\[
\forall x\in\mathsf{BaseIncSeq}(X)\;\exists s\in F\;\bigl(
  s\segs\operatorname{range}(x)\mathbin{\wedge}h(x)=f(s)\bigr).
\]
\end{lemma}
\begin{proof}
The front and value map recorded by $f$ are supplied by
\noderef{SuperSequence}; write $h_F=f.\mathsf{front}$ and
$h_f=f.\mathsf{value}$.  Apply \noderef{SuperSequenceExtension} to
$F$, $X$, $h_F$, and $f$.  We obtain a base multi-sequence $h$, a proof
$h_{\mathrm{lc}}$ of $\mathsf{BaseLocallyConstant}(h)$, and the following
conditional extension equation: whenever $y\in\mathsf{BaseIncSeq}(X)$,
$u$ is a finite set, and
$p:\mathsf{FrontPrefix}(F,u,y.1)$, then
\[
 h(y)=h_f\!\left(\langle u,p.1\rangle\right).                 \tag{1}
\]
The first component $p.1$ is a proof that $u\in F$, by the definition of
\noderef{FrontPrefix}.

We prove the perfectness clause.  Fix an arbitrary
$x\in\mathsf{BaseIncSeq}(X)$.  By the definition of
\noderef{BaseIncSeq}, $x.2$ is the range-containment statement
\[
 \operatorname{range}(x.1)\subseteq X.                         \tag{2}
\]
Apply \noderef{FrontPrefixExists} to $F$, $X$, $h_F$, $x.1$, and (2).
Choose $s_0$ and
$p_0:\mathsf{FrontPrefix}(F,s_0,x.1)$.  The shifted enumeration also has
range contained in $X$: its proof is the second component of
$\mathsf{BaseShift}_X(x)$, whose definition in \noderef{BaseShift} gives
\[
 (\mathsf{BaseShift}_X(x)).1=\mathsf{ShiftMap}(x.1).              \tag{3}
\]
Here we use both the composition formula in \noderef{BaseShift} and the
definition of that composition as \noderef{ShiftMap}.  Apply
\noderef{FrontPrefixExists} again, now to $F$, $X$, $h_F$,
$(\mathsf{BaseShift}_X(x)).1$, and the containment proof
$(\mathsf{BaseShift}_X(x)).2$.  Choose $t_0$ and
$q_0:\mathsf{FrontPrefix}(F,t_0,(\mathsf{BaseShift}_X(x)).1)$.

Unpacking $p_0$ and $q_0$ with \noderef{FrontPrefix}, and using the
definition of \noderef{ProperPrefixSet}, gives
\[
 s_0\in F,\qquad
 \mathsf{ProperPrefixSet}(s_0,\operatorname{range}(x.1)),
\tag{4}
\]
and
\[
 t_0\in F,\qquad
 \mathsf{ProperPrefixSet}
   \bigl(t_0,\operatorname{range}((\mathsf{BaseShift}_X(x)).1)\bigr).
\tag{5}
\]
Using (3), the second assertion in (5) rewrites as
\[
 \mathsf{ProperPrefixSet}
   \bigl(t_0,\operatorname{range}(\mathsf{ShiftMap}(x.1))\bigr).
\tag{6}
\]

Apply the conditional equation (1) directly to $(x,s_0,p_0)$.  This
gives
\[
 e_s:\qquad h(x)=h_f\!\left(\langle s_0,p_0.1\rangle\right).
\tag{7}
\]
Applying (1) directly to
$(\mathsf{BaseShift}_X(x),t_0,q_0)$ gives
\[
 e_t:\qquad h(\mathsf{BaseShift}_X(x))
   =h_f\!\left(\langle t_0,q_0.1\rangle\right).
\tag{8}
\]
The same first equality explicitly supplies the extension witness required
in the conclusion: choose $s=s_0$ and $hs=p_0$ to obtain
\[
 \exists s\;\exists hs:\mathsf{FrontPrefix}(F,s,x.1),\quad
 h(x)=h_f\!\left(\langle s,hs.1\rangle\right).
\tag{9}
\]

We now construct the exact witness required by
\noderef{FiniteShift}.  Take the increasing enumeration $x.1$ itself.
The first conjunct in the definition of $\mathsf{FiniteShift}$ is the
second assertion in (4), and the second conjunct is (6).  Therefore, by
the definition in \noderef{FiniteShift},
\[
 \mathsf{FiniteShift}(s_0,t_0)                                  \tag{10}
\]
holds with witness $x.1$.

The first components $p_0.1$ and $q_0.1$ prove $s_0\in F$ and $t_0\in F$.
Apply the perfectness hypothesis from the lemma to these members and
(10), using the definition in
\noderef{PerfectSuperSequence}.  We obtain
\[
 r\!\left(h_f\!\left(\langle s_0,p_0.1\rangle\right),
          h_f\!\left(\langle t_0,q_0.1\rangle\right)\right).       \tag{11}
\]
Substitute (7) and (8) into (11).  This yields
\[
 r\bigl(h(x),h(\mathsf{BaseShift}_X(x))\bigr).                    \tag{12}
\]
Since $x$ was arbitrary, (12) is precisely
$\mathsf{BasePerfect}(r,h)$ by \noderef{BasePerfect}.  Together with
$h_{\mathrm{lc}}$ and the extension witness (9) for each $x$, this is
exactly the conclusion of the lemma.
\end{proof}
