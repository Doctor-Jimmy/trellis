\begin{lemma}[Restriction preserves local constancy]
If $h$ is a locally constant multi-sequence and $z\in\mathsf{IncSeq}$,
then the restricted multi-sequence $h\mathbin{|}z$ is locally constant.
\end{lemma}
\begin{proof}
Assume that $h$ is locally constant and fix $x\in\mathsf{IncSeq}$.  Put
\[
u=\mathsf{IncSeqComp}(z,x).
\]
By \noderef{LocallyConstant}, there is an integer $m$ such that every
$v\in\mathsf{IncSeq}$ which agrees with $u$ in all coordinates $i<m$
satisfies $h(v)=h(u)$.  By \noderef{IncSeqComp}, the coordinates of $u$ are
\[
u(i)=z(x(i))
\quad\text{for every }i.
\]
The fact that this is an increasing sequence uses that $x$ and $z$ are order
embeddings; in particular, by \noderef{IncSeq}, $i<j$ implies
$x(i)<x(j)$ and then $z(x(i))<z(x(j))$.

Choose the prefix length $n=m$.  Let $y\in\mathsf{IncSeq}$ agree with $x$ in
all coordinates $i<n$, and put
\[
w=\mathsf{IncSeqComp}(z,y).
\]
Again \noderef{IncSeqComp} gives $w(i)=z(y(i))$.  If $i<m=n$, the agreement
of $x$ and $y$ gives $x(i)=y(i)$, so applying the same function $z$ gives
\[
u(i)=z(x(i))=z(y(i))=w(i).
\]
Thus $w$ agrees with $u$ in every coordinate $i<m$, and the deciding property
of $u$ yields $h(w)=h(u)$.  Finally, by
\noderef{MultiSequenceRestrict},
\[
(h\mathbin{|}z)(y)=h(\mathsf{IncSeqComp}(z,y))=h(w)=h(u)
 =h(\mathsf{IncSeqComp}(z,x))=(h\mathbin{|}z)(x).
\]
We have therefore found, for the arbitrary $x$, a length $n$ such that every
$y$ agreeing with $x$ through that length has the same restricted value.
This is precisely local constancy of $h\mathbin{|}z$.
\end{proof}
